from pathlib import Path

directory = Path("argus/cli/commands")
changed = []

for path in directory.glob("*.py"):
    source = path.read_text()

    if "argparse.ArgumentParser" not in source:
        continue

    source = source.replace(
        "argparse.ArgumentParser",
        "cmd2.Cmd2ArgumentParser"
    )

    if "import cmd2" not in source:
        lines = source.splitlines(keepends=True)
        insert_at = 0

        for i, line in enumerate(lines):
            if line.startswith("from __future__ import "):
                insert_at = i + 1
            elif line.strip() and not line.lstrip().startswith("#"):
                break

        lines.insert(insert_at, "import cmd2\n")
        source = "".join(lines)

    path.write_text(source)
    changed.append(str(path))

print("Archivos modificados:")
for path in changed:
    print(f"  {path}")

if not changed:
    print("No se encontraron archivos que modificar.")
PY
